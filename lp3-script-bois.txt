function nextWeeklyDeadline(){

function parseConfigDate(str, defHour){

// dims biaisées vers le petit : min H30/L40/P8 cm, max H140/L120/P30 cm
function skewRand(min,max){ return min + Math.pow(Math.random(),2)*(max-min); }
function fmtCm(v){
v = Math.round(v*2)/2;
return (v%1===0) ? String(v) : String(v).replace('.',',');
}
function randDims(){
var H = fmtCm(skewRand(30,140));
var L = fmtCm(skewRand(40,120));
var P = fmtCm(skewRand(8,30));
return H+"×"+L+"×"+P+" cm";
}
